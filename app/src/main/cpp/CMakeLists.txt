cmake_minimum_required(VERSION 3.22.1)
project(chmreader C CXX)
set(CHMLIB ${CMAKE_CURRENT_SOURCE_DIR}/../../../../third_party/chmlib/src)
add_library(chm SHARED ${CHMLIB}/chm_lib.c ${CHMLIB}/lzx.c)
target_compile_definitions(chm PRIVATE CHM_MT CHM_USE_PREAD CHM_USE_IO64 _FILE_OFFSET_BITS=64)
target_include_directories(chm PUBLIC ${CHMLIB})
target_compile_options(chm PRIVATE -Wno-implicit-function-declaration)
target_link_options(chm PRIVATE -Wl,-z,max-page-size=16384)
add_library(chmbridge SHARED bridge.cpp)
target_link_libraries(chmbridge chm)
target_compile_features(chmbridge PRIVATE cxx_std_17)
target_link_options(chmbridge PRIVATE -Wl,-z,max-page-size=16384)
